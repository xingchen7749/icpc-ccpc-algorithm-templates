## 7.1 线段相交：整数精确判断

[完整代码：7.1 线段相交：整数精确判断.cpp](../%E4%BB%A3%E7%A0%81/07_%E8%AE%A1%E7%AE%97%E5%87%A0%E4%BD%95/7.1%20%E7%BA%BF%E6%AE%B5%E7%9B%B8%E4%BA%A4%EF%BC%9A%E6%95%B4%E6%95%B0%E7%B2%BE%E7%A1%AE%E5%88%A4%E6%96%AD.cpp)

**用途与条件**：判断两条闭线段是否相交，包含端点、共线重叠和退化点。

**复杂度**：时间与额外空间 O(1)。

**易错点**：坐标绝对值要求不超过 10^18；包含端点接触、共线重叠和退化线段，使用整数精确比较。

```cpp

struct Point {
    i64 x, y;
    // 按 (x,y) 字典序比较当前点和 b，供严格排序使用；不使用浮点容差，O(1)。
    bool operator<(Point b) const { return tie(x, y) < tie(b.x, b.y); }
    // 判断当前点与 b 的两个整数坐标是否完全相等，供点集去重使用，O(1)。
    bool operator==(Point b) const { return x == b.x && y == b.y; }
};
// 返回向量 AB 与 AC 的叉积，即三点构成三角形的两倍有向面积。
// 正值表示 a→b→c 逆时针，0 表示共线；坐标绝对值要求不超过 10^18，O(1)。
i128 cross(Point a, Point b, Point c) {
    return (i128(b.x) - a.x) * (i128(c.y) - a.y) - (i128(b.y) - a.y) * (i128(c.x) - a.x);
}
// 返回 x 的符号：负数为 -1，零为 0，正数为 1；时间 O(1)。
int sign(i128 x) { return (x > 0) - (x < 0); }
// 判断点 p 是否在线段 ab 上，包含端点，也支持 a=b 的退化线段。
// 使用精确叉积和坐标范围判断，坐标满足整数几何范围约定；时间 O(1)。
bool on_segment(Point p, Point a, Point b) {
    return cross(a, b, p) == 0 && min(a.x, b.x) <= p.x && p.x <= max(a.x, b.x) &&
           min(a.y, b.y) <= p.y && p.y <= max(a.y, b.y);
}


// 判断闭线段 ab 与 cd 是否有交点，包含端点接触、共线重叠和退化点。
// 使用整数精确判断，坐标绝对值不超过 10^18；时间 O(1)。
bool intersect(Point a, Point b, Point c, Point d) {
    int x = sign(cross(a, b, c)), y = sign(cross(a, b, d));
    int z = sign(cross(c, d, a)), w = sign(cross(c, d, b));
    if (!x && on_segment(c, a, b)) return true;
    if (!y && on_segment(d, a, b)) return true;
    if (!z && on_segment(a, c, d)) return true;
    if (!w && on_segment(b, c, d)) return true;
    return x * y < 0 && z * w < 0;
}
```

**调用与说明**：intersect({0,0},{2,0},{1,0},{3,0})==true。

## 7.2 凸包：单调链

[完整代码：7.2 凸包：单调链.cpp](../%E4%BB%A3%E7%A0%81/07_%E8%AE%A1%E7%AE%97%E5%87%A0%E4%BD%95/7.2%20%E5%87%B8%E5%8C%85%EF%BC%9A%E5%8D%95%E8%B0%83%E9%93%BE.cpp)

**用途与条件**：求整数点集的逆时针凸包，去重且删除边上中间共线点；返回无占位顶点列表。

**复杂度**：时间 O(n log n)，空间 O(n)。

**易错点**：要求坐标绝对值≤10^18，以保证差值及叉积落在数值范围内。排序使用严格字典序，不加 eps。全共线凸包只保留两个端点。

```cpp

struct Point {
    i64 x, y;
    // 按 (x,y) 字典序比较当前点和 b，供严格排序使用；不使用浮点容差，O(1)。
    bool operator<(Point b) const { return tie(x, y) < tie(b.x, b.y); }
    // 判断当前点与 b 的两个整数坐标是否完全相等，供点集去重使用，O(1)。
    bool operator==(Point b) const { return x == b.x && y == b.y; }
};
// 返回向量 AB 与 AC 的叉积，即三点构成三角形的两倍有向面积。
// 正值表示 a→b→c 逆时针，0 表示共线；坐标绝对值要求不超过 10^18，O(1)。
i128 cross(Point a, Point b, Point c) {
    return (i128(b.x) - a.x) * (i128(c.y) - a.y) - (i128(b.y) - a.y) * (i128(c.x) - a.x);
}
// 返回 x 的符号：负数为 -1，零为 0，正数为 1；时间 O(1)。
int sign(i128 x) { return (x > 0) - (x < 0); }
// 判断点 p 是否在线段 ab 上，包含端点，也支持 a=b 的退化线段。
// 使用精确叉积和坐标范围判断，坐标满足整数几何范围约定；时间 O(1)。
bool on_segment(Point p, Point a, Point b) {
    return cross(a, b, p) == 0 && min(a.x, b.x) <= p.x && p.x <= max(a.x, b.x) &&
           min(a.y, b.y) <= p.y && p.y <= max(a.y, b.y);
}


// 返回点集 p 的逆时针凸包顶点列表，不带占位、不重复首点，删除边上中间共线点。
// 输入可无序、有重复，不修改调用者的点集；全共线时只留端点，时间 O(n log n)。
vector<Point> convex_hull(vector<Point> p) {
    sort(p.begin(), p.end());
    p.erase(unique(p.begin(), p.end()), p.end());
    int n = p.size();
    if (n <= 1) return p;
    vector<Point> h(2 * n);
    int k = 0;
    for (auto v : p) {
        while (k >= 2 && cross(h[k - 2], h[k - 1], v) <= 0) {
            --k;
        }
        h[k++] = v;
    }
    for (int i = n - 2, t = k + 1; i >= 0; --i) {
        while (k >= t && cross(h[k - 2], h[k - 1], p[i]) <= 0) {
            --k;
        }
        h[k++] = p[i];
    }
    h.resize(k - 1);
    return h;
}
```

**调用与说明**：convex_hull({{0,0},{1,0},{1,1},{0,1}}) 返回四个逆时针顶点。

## 7.3 多边形面积

[完整代码：7.3 多边形面积.cpp](../%E4%BB%A3%E7%A0%81/07_%E8%AE%A1%E7%AE%97%E5%87%A0%E4%BD%95/7.3%20%E5%A4%9A%E8%BE%B9%E5%BD%A2%E9%9D%A2%E7%A7%AF.cpp)

**用途与条件**：顶点按边界顺序给出；返回两倍有向面积，逆时针为正。

**复杂度**：时间 O(n)，额外空间 O(1)。

**易错点**：点列表无占位、不重复首点；坐标及叉积累加须在数值范围内。

```cpp

struct Point {
    i64 x, y;
    // 按 (x,y) 字典序比较当前点和 b，供严格排序使用；不使用浮点容差，O(1)。
    bool operator<(Point b) const { return tie(x, y) < tie(b.x, b.y); }

    bool operator==(Point b) const { return x == b.x && y == b.y; }
};
// 返回向量 AB 与 AC 的叉积，即三点构成三角形的两倍有向面积。
// 正值表示 a→b→c 逆时针，0 表示共线；坐标绝对值要求不超过 10^18，O(1)。

    return (i128(b.x) - a.x) * (i128(c.y) - a.y) - (i128(b.y) - a.y) * (i128(c.x) - a.x);
}
// 返回 x 的符号：负数为 -1，零为 0，正数为 1；时间 O(1)。
int sign(i128 x) { return (x > 0) - (x < 0); }
// 判断点 p 是否在线段 ab 上，包含端点，也支持 a=b 的退化线段。
// 使用精确叉积和坐标范围判断，坐标满足整数几何范围约定；时间 O(1)。
bool on_segment(Point p, Point a, Point b) {
    return cross(a, b, p) == 0 && min(a.x, b.x) <= p.x && p.x <= max(a.x, b.x) &&
           min(a.y, b.y) <= p.y && p.y <= max(a.y, b.y);
}


// Dependency: Point, cross, on_segment.
// 返回按边界顺序排列的多边形顶点 p 的两倍有向面积，逆时针为正、顺时针为负。
// p 是无占位循环列表，不需重复首点；要求叉积累加不溢出 i128，时间 O(n)。
i128 area2(const vector<Point>& p) {
    i128 a = 0;
    int n = p.size();
    for (int i = 0; i < n; ++i) {
        auto q = p[(i + 1) % n];
        a += i128(p[i].x) * q.y - i128(p[i].y) * q.x;
    }
    return a;
}
```

**调用与说明**：area2({{0,0},{4,0},{4,4},{0,4}})==32。

## 7.4 点与多边形的位置关系

[完整代码：7.4 点与多边形的位置关系.cpp](../%E4%BB%A3%E7%A0%81/07_%E8%AE%A1%E7%AE%97%E5%87%A0%E4%BD%95/7.4%20%E7%82%B9%E4%B8%8E%E5%A4%9A%E8%BE%B9%E5%BD%A2%E7%9A%84%E4%BD%8D%E7%BD%AE%E5%85%B3%E7%B3%BB.cpp)

**用途与条件**：判断点与简单多边形的位置；返回 0 外部、1 边界、2 内部。

**复杂度**：时间 O(n)，额外空间 O(1)。

**易错点**：点列表无占位、不重复首点；坐标及叉积累加须在数值范围内。

```cpp

struct Point {
    i64 x, y;
    // 按 (x,y) 字典序比较当前点和 b，供严格排序使用；不使用浮点容差，O(1)。
    bool operator<(Point b) const { return tie(x, y) < tie(b.x, b.y); }
    // 判断当前点与 b 的两个整数坐标是否完全相等，供点集去重使用，O(1)。
    bool operator==(Point b) const { return x == b.x && y == b.y; }
};
// 返回向量 AB 与 AC 的叉积，即三点构成三角形的两倍有向面积。
// 正值表示 a→b→c 逆时针，0 表示共线；坐标绝对值要求不超过 10^18，O(1)。
i128 cross(Point a, Point b, Point c) {
    return (i128(b.x) - a.x) * (i128(c.y) - a.y) - (i128(b.y) - a.y) * (i128(c.x) - a.x);
}
// 返回 x 的符号：负数为 -1，零为 0，正数为 1；时间 O(1)。
int sign(i128 x) { return (x > 0) - (x < 0); }
// 判断点 p 是否在线段 ab 上，包含端点，也支持 a=b 的退化线段。
// 使用精确叉积和坐标范围判断，坐标满足整数几何范围约定；时间 O(1)。
bool on_segment(Point p, Point a, Point b) {
    return cross(a, b, p) == 0 && min(a.x, b.x) <= p.x && p.x <= max(a.x, b.x) &&
           min(a.y, b.y) <= p.y && p.y <= max(a.y, b.y);
}


// 判断点 q 与简单多边形 p 的位置关系：0 为外部、1 为边界、2 为内部。
// p 是按边界顺序排列的无占位循环列表，顺逆时针均可；时间 O(n)。
int inside(const vector<Point>& p, Point q) {
    // wn（winding number）：绕数；点不在边界时，非零表示在多边形内部。
    int wn = 0, n = p.size();
    for (int i = 0; i < n; ++i) {
        Point a = p[i], b = p[(i + 1) % n];
        if (on_segment(q, a, b)) return 1;
        if (a.y <= q.y && b.y > q.y && cross(a, b, q) > 0) {
            ++wn;
        }
        if (a.y > q.y && b.y <= q.y && cross(a, b, q) < 0) {
            --wn;
        }
    }
    return wn ? 2 : 0;
}
```

**调用与说明**：inside({{0,0},{4,0},{4,4},{0,4}},{2,2})==2。

## 7.5 旋转卡壳：凸包直径

[完整代码：7.5 旋转卡壳：凸包直径.cpp](../%E4%BB%A3%E7%A0%81/07_%E8%AE%A1%E7%AE%97%E5%87%A0%E4%BD%95/7.5%20%E6%97%8B%E8%BD%AC%E5%8D%A1%E5%A3%B3%EF%BC%9A%E5%87%B8%E5%8C%85%E7%9B%B4%E5%BE%84.cpp)

**用途与条件**：求逆时针凸多边形最远点对距离平方；不足两个点返回 0。

**复杂度**：时间 O(n)，额外空间 O(1)。

**易错点**：输入必须逆时针且去除边上中间共线点，可用凸包模板预处理；坐标绝对值不超过 10^18。

```cpp

struct Point {
    i64 x, y;
    // 按 (x,y) 字典序比较当前点和 b，供严格排序使用；不使用浮点容差，O(1)。
    bool operator<(Point b) const { return tie(x, y) < tie(b.x, b.y); }
    // 判断当前点与 b 的两个整数坐标是否完全相等，供点集去重使用，O(1)。
    bool operator==(Point b) const { return x == b.x && y == b.y; }
};
// 返回向量 AB 与 AC 的叉积，即三点构成三角形的两倍有向面积。
// 正值表示 a→b→c 逆时针，0 表示共线；坐标绝对值要求不超过 10^18，O(1)。
i128 cross(Point a, Point b, Point c) {
    return (i128(b.x) - a.x) * (i128(c.y) - a.y) - (i128(b.y) - a.y) * (i128(c.x) - a.x);
}
// 返回 x 的符号：负数为 -1，零为 0，正数为 1；时间 O(1)。
int sign(i128 x) { return (x > 0) - (x < 0); }
// 判断点 p 是否在线段 ab 上，包含端点，也支持 a=b 的退化线段。
// 使用精确叉积和坐标范围判断，坐标满足整数几何范围约定；时间 O(1)。
bool on_segment(Point p, Point a, Point b) {
    return cross(a, b, p) == 0 && min(a.x, b.x) <= p.x && p.x <= max(a.x, b.x) &&
           min(a.y, b.y) <= p.y && p.y <= max(a.y, b.y);
}


// 返回整数点 a、b 的欧氏距离平方，避免开方；坐标绝对值不超过 10^18，时间 O(1)。
i128 dist2(Point a, Point b) {
    i128 x = i128(a.x) - b.x, y = i128(a.y) - b.y;
    return x * x + y * y;
}
// 用旋转卡壳返回凸多边形最远两点距离的平方；不足两个点返回 0。
// p 必须逆时针排列且无边上中间共线点，可直接使用 convex_hull 的结果；时间 O(n)。
i128 diameter2(const vector<Point>& p) {
    int n = p.size();
    if (n < 2) return 0;
    if (n == 2) return dist2(p[0], p[1]);
    i128 ans = 0;
    int j = 1;
    for (int i = 0; i < n; ++i) {
        int ni = (i + 1) % n;
        while (cross(p[i], p[ni], p[(j + 1) % n]) > cross(p[i], p[ni], p[j])) {
            j = (j + 1) % n;
        }
        ans = max({ans, dist2(p[i], p[j]), dist2(p[ni], p[j])});
    }
    return ans;
}
```

**调用与说明**：diameter2({{0,0},{1,0},{1,1},{0,1}})==2。

## 7.6 直线交点

[完整代码：7.6 直线交点.cpp](../%E4%BB%A3%E7%A0%81/07_%E8%AE%A1%E7%AE%97%E5%87%A0%E4%BD%95/7.6%20%E7%9B%B4%E7%BA%BF%E4%BA%A4%E7%82%B9.cpp)

**用途与条件**：直线由点和非零方向向量表示；返回唯一交点，平行或重合返回 nullopt。

**复杂度**：时间和空间 O(1)。

**易错点**：方向向量必须非零；EPS 为绝对容差，应按坐标尺度调整。平行和重合都返回 nullopt。

```cpp

// ld（long double）：扩展精度浮点类型；P（point）：二维点，也用于表示向量。
using ld = long double;
// EPS（epsilon）：浮点比较容差，需结合题目坐标尺度选择。
const ld EPS = 1e-12L;
struct P {
    ld x, y;
    // 返回当前二维向量与 b 的逐坐标和，不修改原向量；时间 O(1)。
    P operator+(P b) const { return {x + b.x, y + b.y}; }
    // 返回当前二维向量减去 b 的结果，可表示从 b 指向当前点的位移；时间 O(1)。
    P operator-(P b) const { return {x - b.x, y - b.y}; }
    // 返回当前向量乘标量 k 的结果，不修改原向量；时间 O(1)。
    P operator*(ld k) const { return {x * k, y * k}; }
};
// 返回二维向量 a、b 的叉积，正负表示旋转方向；浮点结果需结合 EPS 判断，O(1)。
ld cross(P a, P b) { return a.x * b.y - a.y * b.x; }
// 返回二维向量 a、b 的点积，可用于投影和长度平方计算；时间 O(1)。
ld dot(P a, P b) { return a.x * b.x + a.y * b.y; }


// 返回直线 p+t*v 与 q+s*w 的唯一交点；v、w 必须是非零方向向量。
// 在 EPS 意义下平行或重合时返回 nullopt，不区分这两种情况；时间 O(1)。
optional<P> line_intersection(P p, P v, P q, P w) {
    ld d = cross(v, w);
    if (fabsl(d) < EPS) return nullopt;
    return p + v * (cross(q - p, w) / d);
}
```

**调用与说明**：line_intersection({0,0},{1,1},{0,1},{1,-1}) 为 (0.5,0.5)。直线平行和重合均返回无单一交点。

## 7.7 圆与圆交点

[完整代码：7.7 圆与圆交点.cpp](../%E4%BB%A3%E7%A0%81/07_%E8%AE%A1%E7%AE%97%E5%87%A0%E4%BD%95/7.7%20%E5%9C%86%E4%B8%8E%E5%9C%86%E4%BA%A4%E7%82%B9.cpp)

**用途与条件**：两圆半径非负；返回交点数及交点列表，-1 表示重合且交点无穷。

**复杂度**：时间和空间 O(1)。

**易错点**：半径必须非负；EPS 为绝对容差，应按坐标尺度调整。重合的零半径圆只有一个交点。

```cpp

// ld（long double）：扩展精度浮点类型；P（point）：二维点，也用于表示向量。
using ld = long double;
// EPS（epsilon）：浮点比较容差，需结合题目坐标尺度选择。
const ld EPS = 1e-12L;
struct P {
    ld x, y;
    // 返回当前二维向量与 b 的逐坐标和，不修改原向量；时间 O(1)。
    P operator+(P b) const { return {x + b.x, y + b.y}; }
    // 返回当前二维向量减去 b 的结果，可表示从 b 指向当前点的位移；时间 O(1)。
    P operator-(P b) const { return {x - b.x, y - b.y}; }
    // 返回当前向量乘标量 k 的结果，不修改原向量；时间 O(1)。
    P operator*(ld k) const { return {x * k, y * k}; }
};
// 返回二维向量 a、b 的叉积，正负表示旋转方向；浮点结果需结合 EPS 判断，O(1)。
ld cross(P a, P b) { return a.x * b.y - a.y * b.x; }
// 返回二维向量 a、b 的点积，可用于投影和长度平方计算；时间 O(1)。
ld dot(P a, P b) { return a.x * b.x + a.y * b.y; }


// 求圆心 a、半径 r 与圆心 b、半径 s 两圆的交点，要求半径非负。
// 返回 {数量,交点列表}：0/1/2 为有限交点数，-1 表示重合且有无穷交点。
// 按 EPS 判定，重合的零半径圆按一个交点处理；时间和空间 O(1)。
pair<int, vector<P>> circle_intersection(P a, ld r, P b, ld s) {
    assert(r >= 0 && s >= 0);
    P v = b - a;
    ld d = sqrtl(dot(v, v));
    if (d < EPS) {
        if (fabsl(r - s) > EPS) return {0, {}};
        if (r < EPS) return {1, {a}};
        return {-1, {}};
    }
    if (d > r + s + EPS || d < fabsl(r - s) - EPS) return {0, {}};
    ld x = (r * r - s * s + d * d) / (2 * d), h2 = r * r - x * x;
    if (h2 < -EPS) return {0, {}};
    P base = a + v * (x / d);
    if (h2 <= EPS) return {1, {base}};
    P off = P{-v.y, v.x} * (sqrtl(max((ld)0, h2)) / d);
    return {2, {base + off, base - off}};
}
```

**调用与说明**：circle_intersection({0,0},1,{2,0},1).first==1。

## 7.8 有界半平面交：逐边裁剪

[完整代码：7.8 有界半平面交：逐边裁剪.cpp](../%E4%BB%A3%E7%A0%81/07_%E8%AE%A1%E7%AE%97%E5%87%A0%E4%BD%95/7.8%20%E6%9C%89%E7%95%8C%E5%8D%8A%E5%B9%B3%E9%9D%A2%E4%BA%A4%EF%BC%9A%E9%80%90%E8%BE%B9%E8%A3%81%E5%89%AA.cpp)

**用途与条件**：用凸多边形逐个裁剪半平面，保留有向直线 a→b 左侧（含边界）。依赖浮点几何 P/cross/EPS。 例外：poly 是无占位的循环顶点列表，位置从 0 开始，与本章其他几何函数一致。

**复杂度**：单次 O(当前顶点数)，m 个半平面总体最坏 O(m²)；空间 O(m)。

**易错点**：这是适合中小规模的裁剪版，不是 O(m log m) 排序双端队列版。必须给出题目保证覆盖解集的初始包围框；任意猜大框不能可靠判断无界。

```cpp

// ld（long double）：扩展精度浮点类型；P（point）：二维点，也用于表示向量。
using ld = long double;
// EPS（epsilon）：浮点比较容差，需结合题目坐标尺度选择。
const ld EPS = 1e-12L;
struct P {
    ld x, y;
    // 返回当前二维向量与 b 的逐坐标和，不修改原向量；时间 O(1)。
    P operator+(P b) const { return {x + b.x, y + b.y}; }
    // 返回当前二维向量减去 b 的结果，可表示从 b 指向当前点的位移；时间 O(1)。
    P operator-(P b) const { return {x - b.x, y - b.y}; }
    // 返回当前向量乘标量 k 的结果，不修改原向量；时间 O(1)。
    P operator*(ld k) const { return {x * k, y * k}; }
};
// 返回二维向量 a、b 的叉积，正负表示旋转方向；浮点结果需结合 EPS 判断，O(1)。
ld cross(P a, P b) { return a.x * b.y - a.y * b.x; }
// 返回二维向量 a、b 的点积，可用于投影和长度平方计算；时间 O(1)。
ld dot(P a, P b) { return a.x * b.x + a.y * b.y; }


// Dependency: P, cross, EPS.
// 用有向直线 a→b 裁剪凸多边形 poly，保留其左侧及边界，返回新的循环顶点列表。
// 要求 a!=b，顶点按边界顺序排列且不带占位；空结果表示在容差意义下无交集。
// 不修改输入，每次耗时 O(顶点数)；连续裁剪需从保证覆盖解集的包围框开始。
vector<P> clip_halfplane(const vector<P>& poly, P a, P b) {
    vector<P> out;
    int n = poly.size();
    P v = b - a;
    assert(dot(v, v) > 0);
    for (int i = 0; i < n; ++i) {
        P p = poly[i], q = poly[(i + 1) % n];
        // dp、dq：点 p、q 相对裁剪直线的有向叉积值，此处 dp 不是动态规划。
        ld dp = cross(v, p - a), dq = cross(v, q - a);
        // ip、iq：对应端点是否位于保留半平面内，包含容差范围内的边界。
        bool ip = dp >= -EPS, iq = dq >= -EPS;
        if (ip) out.push_back(p);
        if (ip != iq) {
            // Classification uses EPS; clamp near-boundary interpolation.
            ld t = dp / (dp - dq);
            t = max((ld)0, min((ld)1, t));
            out.push_back(p + (q - p) * t);
        }
    }
    return out;
}
```

**调用与说明**：初始矩形按逆时针给出。每条限制调用 poly=clip_halfplane(poly,a,b)；空数组表示交集为空（在浮点容差意义下）。
