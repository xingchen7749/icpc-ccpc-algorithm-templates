## 2.1 并查集

[完整代码：2.1 并查集.cpp](../%E4%BB%A3%E7%A0%81/02_%E6%95%B0%E6%8D%AE%E7%BB%93%E6%9E%84/2.1%20%E5%B9%B6%E6%9F%A5%E9%9B%86.cpp)

**用途与条件**：维护无向连通分量，递归路径压缩配合按秩合并。使用全局 father、rk、sz 数组和 init、find、merge、size 函数，size(id) 查询所在集合的元素数量。元素编号为 1..n，0 号位置不使用。

**复杂度**：均摊 O(α(n)) 每次操作；空间 O(n)。

**易错点**：每组数据先调用 init(n)。rk 表示秩，路径压缩后不等于实际树高，也不是集合大小。只有根节点的 sz 才表示当前集合大小，应通过 size(id) 查询。merge 成功时可在调用处将连通块数量减一。普通并查集不能删除边；回滚版本不要使用路径压缩。

```cpp
// DSU（Disjoint Set Union）：并查集。
// father[x]：父节点；根节点的父节点是自身。
// rk（rank）：合并使用的秩，不是集合大小；sz[root]：根对应的集合大小。
vector<int> father, rk, sz;

// 初始化编号 1..n 的独立集合，初始秩和集合大小均为 1；时间和空间 O(n)。
void init(int n) {
    father.resize(n + 1);
    rk.assign(n + 1, 1);
    sz.assign(n + 1, 1);
    for (int i = 1; i <= n; ++i) father[i] = i;
}

// 返回 id 所在集合的根，并通过递归压缩路径；要求 1<=id<=n。
int find(int id) {
    if (id != father[id]) father[id] = find(father[id]);
    return father[id];
}

// 按秩合并 u、v 所在集合并更新集合大小；成功返回 true，原本同集合返回 false。
// 要求顶点为 1..n；配合路径压缩，均摊时间 O(α(n))。
bool merge(int u, int v) {
    int a = find(u), b = find(v);
    if (a == b) return false;
    if (rk[a] > rk[b]) {
        father[b] = a;
        sz[a] += sz[b];
    }
    else if (rk[b] > rk[a]) {
        father[a] = b;
        sz[b] += sz[a];
    }
    else {
        father[b] = a;
        rk[a]++;
        sz[a] += sz[b];
    }
    return true;
}

// 返回 id 所在集合的元素数量；要求 1<=id<=n，均摊时间 O(α(n))。
int size(int id) {
    return sz[find(id)];
}
```

**调用与说明**：init(n); int num=n; 若 merge(u,v) 返回 true，则 --num；find(u)==find(v) 判断是否连通；size(u) 查询 u 所在集合的元素数量。

## 2.2 可回滚并查集

[完整代码：2.2 可回滚并查集.cpp](../%E4%BB%A3%E7%A0%81/02_%E6%95%B0%E6%8D%AE%E7%BB%93%E6%9E%84/2.2%20%E5%8F%AF%E5%9B%9E%E6%BB%9A%E5%B9%B6%E6%9F%A5%E9%9B%86.cpp)

**用途与条件**：离线动态连通性、分治时间轴等场景。保存快照，在处理完一个分支后恢复。元素编号为 1..n，0 号位置不使用。

**复杂度**：find / merge O(log n)；回滚每次成功合并 O(1)；空间 O(n+操作数)。

**易错点**：禁止路径压缩。失败的 merge 不入栈，所以应记录 snapshot 而非假定每次操作都对应一条历史。

```cpp
// DSU（Disjoint Set Union）：并查集；Rollback 表示支持回滚。
struct RollbackDSU {
    // father[x]：父节点；sz[root]：根对应的集合大小。
    vector<int> father, sz;
    // hist（history）：成功合并的历史，保存被挂接的根及合并前另一集合的大小。
    vector<pair<int, int>> hist;
    // 创建编号 1..n 的独立集合并清空合并历史，时间和空间 O(n)。
    RollbackDSU(int n) : father(n + 1), sz(n + 1, 1) {
        iota(father.begin(), father.end(), 0);
    }
    // 沿父指针返回 x 的代表元，不压缩路径以便回滚；时间 O(log n)。
    int find(int x) {
        while (x != father[x]) x = father[x];
        return x;
    }
    // 返回当前成功合并历史的长度，作为之后 rollback 的快照；不修改状态，O(1)。
    int snapshot() const { return hist.size(); }
    // 按大小合并 a、b 所在集合，并记录可撤销信息；同集合返回 false 且不记历史。
    // 有效合并返回 true；时间 O(log n)。
    bool merge(int a, int b) {
        a = find(a);
        b = find(b);
        if (a == b) return false;
        if (sz[a] < sz[b]) swap(a, b);
        hist.push_back({b, sz[a]});
        father[b] = a;
        sz[a] += sz[b];
        return true;
    }
    // 撤销当前历史末尾的合并，恢复到快照 s；s 必须是当前历史中的有效长度。
    // 每撤销一次成功合并耗时 O(1)，不会恢复已被丢弃的未来快照。
    void rollback(int s) {
        while ((int)hist.size() > s) {
            auto [b, old] = hist.back();
            hist.pop_back();
            sz[father[b]] = old;
            father[b] = b;
        }
    }
};
```

**调用与说明**：int s=d.snapshot(); d.merge(u,v); ...; d.rollback(s);

## 2.3 树状数组与第 k 小

[完整代码：2.3 树状数组与第 k 小.cpp](../%E4%BB%A3%E7%A0%81/02_%E6%95%B0%E6%8D%AE%E7%BB%93%E6%9E%84/2.3%20%E6%A0%91%E7%8A%B6%E6%95%B0%E7%BB%84%E4%B8%8E%E7%AC%AC%20k%20%E5%B0%8F.cpp)

**用途与条件**：单点加、闭区间 [l,r] 求和；下标 1..n，sum(r) 为 [1,r] 之和，sum(0)=0。非负频数时 kth(k) 返回第 k 小的 1 下标。

**复杂度**：更新、查询、第 k 小均 O(log n)；空间 O(n)。

**易错点**：kth 要求 1≤k≤总频数，且所有单点值非负。区间和可存负数，但此时不能使用 kth。

```cpp
struct Fenwick {
    int n;
    // t[i]：树状数组节点，维护原数组区间 [i-lowbit(i)+1,i] 的和。
    // lowbit(i) = i & -i，表示 i 的最低有效二进制位对应的值。
    vector<i64> t;
    // 初始化长度为 n、下标为 1..n 的全零树状数组，时间和空间 O(n)。
    Fenwick(int n) : n(n), t(n + 1) {}
    // 将第 x 项增加 v，并更新树状数组；要求 1<=x<=n，时间 O(log n)。
    void add(int x, i64 v) {
        assert(1 <= x && x <= n);
        for (; x <= n; x += x & -x) {
            t[x] += v;
        }
    }
    // 返回闭区间 [1,r] 的元素和；要求 0<=r<=n，r=0 返回 0，时间 O(log n)。
    i64 sum(int r) const {
        i64 s = 0;
        for (; r; r -= r & -r) {
            s += t[r];
        }
        return s;
    }
    // 返回闭区间 [l,r] 的元素和；非空查询要求 1<=l<=r<=n，l>r 返回 0。
    // 不修改数组，时间 O(log n)。
    i64 query(int l, int r) const { return l > r ? 0 : sum(r) - sum(l - 1); }
    // 返回最小的下标 x，使前缀和 sum(x)>=k；将单点值看作频数时即第 k 小的位置。
    // 要求所有单点频数非负且 1<=k<=sum(n)，返回 1 下标，时间 O(log n)。
    int kth(i64 k) const {
        assert(1 <= k && k <= sum(n));
        int x = 0, step = 1;
        while (step <= n / 2) step <<= 1;
        for (; step; step >>= 1) {
            int y = x + step;
            if (y <= n && t[y] < k) {
                x = y;
                k -= t[y];
            }
        }
        return x + 1;
    }
};
```

**调用与说明**：Fenwick f(4); f.add(3,3); f.query(2,4)==3，f.kth(2)==3。逆序对：依次累加已插入数量 - sum(rank)，然后 add(rank,1)。

## 2.4 ST 表：静态区间最小值

[完整代码：2.4 ST 表：静态区间最小值.cpp](../%E4%BB%A3%E7%A0%81/02_%E6%95%B0%E6%8D%AE%E7%BB%93%E6%9E%84/2.4%20ST%20%E8%A1%A8%EF%BC%9A%E9%9D%99%E6%80%81%E5%8C%BA%E9%97%B4%E6%9C%80%E5%B0%8F%E5%80%BC.cpp)

**用途与条件**：静态数组 a[1..n] 的非空闭区间 [l,r] 最小值。a[0] 占位；st[i][j] 保存从 i 开始、长度为 2^j 的区间最小值。

**复杂度**：预处理 O(n log n)，查询 O(1)，空间 O(n log n)。

**易错点**：输入至少包含 a[0] 占位和一个有效元素。构造时断言 a.size() >= 2，查询时断言 1 <= l <= r <= n；定义 NDEBUG 后断言禁用，调用者仍须保证这些条件。数组修改后需要重建。log2 的参数必须为正数。默认数据类型为 int。

```cpp
// ST（Sparse Table）：稀疏表，用于静态区间查询。
// RMQ（Range Minimum Query）：区间最小值查询，本模板默认实现的操作。
class ST {
private:
    int n;
    // st[i][j]：从 i 开始、长度为 2^j 的区间最小值。
    vector<vector<int>> st;

public:
    // 由 a[1..n] 构造静态区间最小值表，忽略 a[0]，要求 n>=1。
    // 预处理时间和空间 O(n log n)；原数组修改后需重新构造。
    ST(const vector<int>& a) {
        assert(a.size() >= 2);
        n = (int)a.size() - 1;
        if (n <= 0) return;

        int K = log2((double)n) + 1;
        st.assign(n + 1, vector<int>(K));

        for (int i = 1; i <= n; ++i) st[i][0] = a[i];

        for (int j = 1; j < K; ++j) {
            for (int i = 1; i + (1LL << j) - 1 <= n; ++i) {
                st[i][j] = min(st[i][j - 1], st[i + (1LL << (j - 1))][j - 1]);
            }
        }
    }

    // 1 下标，查询非空闭区间 [l, r] 的最小值
    int query(int l, int r) const {
        assert(1 <= l && l <= r && r <= n);
        int j = log2((double)(r - l + 1));
        return min(st[l][j], st[r - (1LL << j) + 1][j]);
    }
};
```

**调用与说明**：vector<int> a = {0,3,1,4}; ST st(a); st.query(1,2) 返回 1。独立算法文件中直接使用 ST。



### 更换区间操作

需同时替换两处合并操作，其他建表和查询逻辑不变：

1. 构造函数中，`st[i][j] = min(...)` 的 `min`。
2. `query` 函数中，`return min(...)` 的 `min`。

| 目标操作 | 两处替换方式 |
|---|---|
| 最大值 | 将 `min` 改为 `max` |
| 最大公约数（非负整数） | 将 `min` 改为 `std::gcd`，使用 C++17 |
| 按位与 | 将两处 `min(x, y)` 改为 `(x & y)`，x、y 代表原来的两个参数 |
| 按位或 | 将两处 `min(x, y)` 改为 `(x \| y)`，x、y 代表原来的两个参数 |

更换后同步修改查询注释。合并操作须满足结合律和幂等性（合并 x 与 x 仍为 x），因为查询用的两个区间可能重叠。不能直接替换为求和、乘积或异或。

数据超出 int 范围时，可在标准头文件和 `using namespace std;` 之后、此类及输入数组定义之前添加 `#define int long long`，主函数使用 `signed main()`。使用完整的 `代码/02_数据结构/2.4 ST 表：静态区间最小值.cpp` 写题时，在标准头文件之后取消整数宏的注释即可；此用法针对本 ST 模板。

### 2.5.1 带懒标记线段树（segmenttree）

[完整代码：2.5.1 带懒标记线段树（segmenttree）.cpp](../%E4%BB%A3%E7%A0%81/02_%E6%95%B0%E6%8D%AE%E7%BB%93%E6%9E%84/2.5.1%20%E5%B8%A6%E6%87%92%E6%A0%87%E8%AE%B0%E7%BA%BF%E6%AE%B5%E6%A0%91%EF%BC%88segmenttree%EF%BC%89.cpp)



两版只需 `#include <bits/stdc++.h>` 和 `using namespace std;`，类名直接使用，无命名空间。两版分别位于 `代码/02_数据结构/2.5.1 带懒标记线段树（segmenttree）.cpp` 和 `代码/02_数据结构/2.5.2 无懒标记线段树（SegTree）.cpp`，每份文件只包含对应的类，均带完整起手式，可单独复制写题。需要 64 位时先包含标准头文件，再写 `#define int long long`，再放类和 `vector<int>` 输入数组，主函数用 `signed main()`；不要在宏之后首次包含标准头文件。

只需要单点修改时选择 SegTree；需要区间加时选择 segmenttree。两个版本的 add 参数不同：前者 add(pos,v)，后者 add(l,r,v)。


**两个版本的区别**

| 对比项 | segmenttree（带懒标记） | SegTree（无懒标记） |
|---|---|---|
| 区间查询 | query(l,r)，区间和 | query(l,r)，区间和 |
| 单点赋值 | set(pos,v) | set(pos,v) |
| 加法修改 | add(l,r,v)，整个区间加 v | add(pos,v)，单个元素加 v |
| 懒标记 | lazy 保存已计入当前节点、尚未下传的增量 | 没有 lazy 和 pushdown |
| 使用场景 | 需要区间加 | 只需要单点修改，代码更简单 |

两版建树 O(n)、空间 O(n)，各自提供的修改和查询均为 O(log n)。segmenttree 可以用 add(pos,pos,v) 实现单点加；SegTree 若循环进行 k 次单点加，需 O(k log n)，不能当成高效区间加。



**用途与条件**：输入 a[1..n]，a[0] 占位，n≥1。内外均为闭区间，要求 1≤l≤r≤n。支持区间加、单点赋值、区间求和。

**复杂度**：建树 O(n)，每次操作 O(log n)，空间 O(n)。

**易错点**：所有类型统一 int；区间和、增量乘长度和中间结果都必须在类型范围内。需要时在标准头文件之后、类定义之前添加 #define int long long。lazy 只记录每个元素的增量，不乘长度；单点赋值前下传，叶子赋值后清空标记。构造函数和公开接口使用 assert 检查数组非空、单点下标及区间范围；定义 NDEBUG 后断言禁用，调用方仍须保证参数合法。

```cpp
class segmenttree {
private:
    int n;
    vector<int> tree, lazy; // 区间和、尚未下传的加法标记

    // 合并两个区间的结果：参数 a、b 为各自区间和，返回合并后的和
    int merge(int a, int b) { return a + b; }

    // 向上合并：通过 merge 用两个子节点的结果更新节点 p
    void pushup(int p) {
        tree[p] = merge(tree[p * 2], tree[p * 2 + 1]);
    }

    // 建树：节点 p 管理闭区间 [l,r]
    void build(const vector<int>& a, int p, int l, int r) {
        if (l == r) {
            tree[p] = a[l];
            return;
        }
        int m = (l + r) / 2;
        build(a, p * 2, l, m);
        build(a, p * 2 + 1, m + 1, r);
        pushup(p);
    }

    // 下传标记：更新孩子的区间和与标记，再清空当前标记
    void pushdown(int p, int l, int r) {
        if (l == r || lazy[p] == 0) return;
        int m = (l + r) / 2;
        tree[p * 2] += lazy[p] * (m - l + 1);
        tree[p * 2 + 1] += lazy[p] * (r - m);
        lazy[p * 2] += lazy[p];
        lazy[p * 2 + 1] += lazy[p];
        lazy[p] = 0;
    }

    // 单点赋值的递归实现：将 a[pos] 改为 v
    void set(int p, int l, int r, int pos, int v) {
        if (l == r) {
            tree[p] = v;
            lazy[p] = 0;
            return;
        }
        pushdown(p, l, r);
        int m = (l + r) / 2;
        if (pos <= m) {
            set(p * 2, l, m, pos, v);
        }
        else {
            set(p * 2 + 1, m + 1, r, pos, v);
        }
        pushup(p);
    }

    // 区间加的递归实现：将 [ql,qr] 内每个数增加 v
    void add(int p, int l, int r, int ql, int qr, int v) {
        if (ql <= l && r <= qr) {
            tree[p] += v * (r - l + 1);
            lazy[p] += v;
            return;
        }
        pushdown(p, l, r);
        int m = (l + r) / 2;
        if (ql <= m) {
            add(p * 2, l, m, ql, qr, v);
        }
        if (qr > m) {
            add(p * 2 + 1, m + 1, r, ql, qr, v);
        }
        pushup(p);
    }

    // 区间求和的递归实现：完全覆盖直接返回，否则累加子区间答案
    int query(int p, int l, int r, int ql, int qr) {
        if (ql <= l && r <= qr) return tree[p];
        pushdown(p, l, r);
        int m = (l + r) / 2;
        int sum = 0;
        if (ql <= m) {
            sum = merge(sum, query(p * 2, l, m, ql, qr));
        }
        if (qr > m) {
            sum = merge(sum, query(p * 2 + 1, m + 1, r, ql, qr));
        }
        return sum;
    }

public:
    // 构造函数：a[0] 占位，要求 n >= 1，建树 O(n)
    segmenttree(const vector<int>& a) {
        assert(a.size() >= 2); // 至少包含占位和一个有效元素
        n = (int)a.size() - 1;
        tree.assign(4 * n + 5, 0);
        lazy.assign(4 * n + 5, 0);
        build(a, 1, 1, n);
    }

    // 单点赋值：a[pos] = v，要求 1 <= pos <= n，O(log n)
    void set(int pos, int v) {
        assert(1 <= pos && pos <= n);
        set(1, 1, n, pos, v);
    }

    // 区间加：[l,r] 内每个数增加 v，要求 1 <= l <= r <= n，O(log n)
    void add(int l, int r, int v) {
        assert(1 <= l && l <= r && r <= n);
        add(1, 1, n, l, r, v);
    }

    // 区间求和：返回 [l,r] 的和，要求 1 <= l <= r <= n，O(log n)
    int query(int l, int r) {
        assert(1 <= l && l <= r && r <= n);
        return query(1, 1, n, l, r);
    }
};
```

**调用与说明**：segmenttree s({0,1,2,3}); s.add(1,2,4); s.set(2,10); s.query(1,3) 返回 18。


#### 改为区间最大值或最小值


以本节的求和代码为起点，最大值和最小值版本分别按下表修改；应在建树前改好代码，再重新构造线段树。

| 修改位置 | 区间和（当前） | 区间最大值 | 区间最小值 |
|---|---|---|---|
| merge 的返回值 | return a + b | return max(a,b) | return min(a,b) |
| 内部 query 的初始答案 | int sum = 0 | int sum = numeric_limits<int>::lowest() | int sum = numeric_limits<int>::max() |
| pushup 与 query 的合并调用 | merge(a,b) | 保持不变 | 保持不变 |

pushup 和 query 已统一调用 merge，合并调用不用改；只需改 merge 的返回值和 query 的初始答案。变量 sum 可重命名为 ans。不能将最大值查询的初始答案设为 0，因为元素可能全为负数。numeric_limits<int> 会随 #define int long long 使用对应类型的边界。

除上表外，还要改区间 add 和 pushdown，共四个函数。

区间 add 的完全覆盖分支，从 `tree[p] += v * (r-l+1);` 改为：

```cpp
tree[p] += v;
lazy[p] += v;
```

pushdown 改为以下写法，最大值和最小值版本相同：

```cpp
// 下传区间加标记：所有元素加 v，最大值和最小值也都加 v
void pushdown(int p, int l, int r)
{
    if (l == r || lazy[p] == 0) return;
    tree[p * 2] += lazy[p];
    tree[p * 2 + 1] += lazy[p];
    lazy[p * 2] += lazy[p];
    lazy[p * 2 + 1] += lazy[p];
    lazy[p] = 0;
}
```

build、单点 set、构造函数和公开接口不变。懒标记始终记录每个元素的增量；只有求和时，更新节点值才需要乘区间长度。若已改成最大值，再切换成最小值，只需改 merge 中的 max/min 及 query 的答案初始值。

### 2.5.2 无懒标记线段树（SegTree）

[完整代码：2.5.2 无懒标记线段树（SegTree）.cpp](../%E4%BB%A3%E7%A0%81/02_%E6%95%B0%E6%8D%AE%E7%BB%93%E6%9E%84/2.5.2%20%E6%97%A0%E6%87%92%E6%A0%87%E8%AE%B0%E7%BA%BF%E6%AE%B5%E6%A0%91%EF%BC%88SegTree%EF%BC%89.cpp)

**用途与条件**：输入 a[1..n]，a[0] 占位，n≥1。支持单点赋值、单点加和闭区间求和；单点下标要求 1≤pos≤n，查询要求 1≤l≤r≤n。

**复杂度**：建树 O(n)，每次操作 O(log n)，空间 O(n)。

**易错点**：所有类型统一 int，需要时在标准头文件之后、类定义之前添加 #define int long long。构造函数和公开接口包含 assert 检查，定义 NDEBUG 后禁用。没有懒标记，add(pos,v) 只修改一个元素。

```cpp
class SegTree {
private:
    int n;
    vector<int> tree; // 区间和

    // 合并两个区间的结果：参数 a、b 为各自区间和，返回合并后的和
    int merge(int a, int b) { return a + b; }

    // 向上合并：通过 merge 用两个子节点的结果更新节点 p
    void pushup(int p) {
        tree[p] = merge(tree[p * 2], tree[p * 2 + 1]);
    }

    // 建树：节点 p 管理闭区间 [l,r]
    void build(const vector<int>& a, int p, int l, int r) {
        if (l == r) {
            tree[p] = a[l];
            return;
        }
        int m = (l + r) / 2;
        build(a, p * 2, l, m);
        build(a, p * 2 + 1, m + 1, r);
        pushup(p);
    }

    // 单点赋值的递归实现：将 a[pos] 改为 v
    void set(int p, int l, int r, int pos, int v) {
        if (l == r) {
            tree[p] = v;
            return;
        }
        int m = (l + r) / 2;
        if (pos <= m) {
            set(p * 2, l, m, pos, v);
        }
        else {
            set(p * 2 + 1, m + 1, r, pos, v);
        }
        pushup(p);
    }

    // 单点加的递归实现：将 a[pos] 增加 v，再向上更新区间和
    void add(int p, int l, int r, int pos, int v) {
        if (l == r) {
            tree[p] += v;
            return;
        }
        int m = (l + r) / 2;
        if (pos <= m) {
            add(p * 2, l, m, pos, v);
        }
        else {
            add(p * 2 + 1, m + 1, r, pos, v);
        }
        pushup(p);
    }

    // 区间求和的递归实现：完全覆盖直接返回，否则累加子区间答案
    int query(int p, int l, int r, int ql, int qr) {
        if (ql <= l && r <= qr) return tree[p];
        int m = (l + r) / 2;
        int sum = 0;
        if (ql <= m) {
            sum = merge(sum, query(p * 2, l, m, ql, qr));
        }
        if (qr > m) {
            sum = merge(sum, query(p * 2 + 1, m + 1, r, ql, qr));
        }
        return sum;
    }

public:
    // 构造函数：a[0] 占位，要求 n >= 1，建树 O(n)
    SegTree(const vector<int>& a) {
        assert(a.size() >= 2); // 至少包含占位和一个有效元素
        n = (int)a.size() - 1;
        tree.assign(4 * n + 5, 0);
        build(a, 1, 1, n);
    }

    // 单点赋值：a[pos] = v，要求 1 <= pos <= n，O(log n)
    void set(int pos, int v) {
        assert(1 <= pos && pos <= n);
        set(1, 1, n, pos, v);
    }

    // 单点加：a[pos] += v，要求 1 <= pos <= n，O(log n)
    void add(int pos, int v) {
        assert(1 <= pos && pos <= n);
        add(1, 1, n, pos, v);
    }

    // 区间求和：返回 [l,r] 的和，要求 1 <= l <= r <= n，O(log n)
    int query(int l, int r) {
        assert(1 <= l && l <= r && r <= n);
        return query(1, 1, n, l, r);
    }
};
```

**调用与说明**：SegTree s({0,1,2,3}); s.add(2,4); s.set(3,10); s.query(1,3) 返回 17。

## 2.6 主席树：静态区间第 k 小

[完整代码：2.6 主席树：静态区间第 k 小.cpp](../%E4%BB%A3%E7%A0%81/02_%E6%95%B0%E6%8D%AE%E7%BB%93%E6%9E%84/2.6%20%E4%B8%BB%E5%B8%AD%E6%A0%91%EF%BC%9A%E9%9D%99%E6%80%81%E5%8C%BA%E9%97%B4%E7%AC%AC%20k%20%E5%B0%8F.cpp)

**用途与条件**：值离散化为 1..m。root[i] 表示前 i 个元素的频数，root[0] 为空版本。查询原数组闭区间 [l,r] 第 k 小，k 从 1 开始。

**复杂度**：每次插入、查询 O(log m)；n 次插入空间 O(n log m)。

**易错点**：m≥1，1≤k≤r-l+1。输入值为 1..m 的秩。内部树递归区间 [L,R) 仅用于分割值域；0 号树节点为空节点。vector 扩容后不能继续使用旧引用。

```cpp
struct PersistentKth {
    struct Node {
        // l、r：左右儿子的节点编号，0 表示空节点；cnt：该值域内的元素数量。
        int l = 0, r = 0, cnt = 0;
    };
    int m;
    // t：可持久化节点池，0 号为空节点。
    vector<Node> t{{}};
    // root[i]：插入前 i 个元素后的版本根；root[0] 表示空前缀。
    vector<int> root{0};
    // 设置离散值域为 1..m，创建空前缀版本 root[0]，要求 m>0。
    PersistentKth(int m) : m(m) {
        assert(m > 0);
    }
    // 内部持久化插入：复制 old 版本在值域 [l,r) 上通往秩 x 的路径，频数加一。
    // 返回新根编号，保留旧版本；要求 l<=x<r，新增节点数和时间均为 O(log m)。
    int insert(int old, int l, int r, int x) {
        int p = t.size();
        t.push_back(t[old]);
        ++t[p].cnt;
        if (r - l > 1) {
            int mid = (l + r) / 2;
            if (x < mid) {
                t[p].l = insert(t[old].l, l, mid, x);
            }
            else {
                t[p].r = insert(t[old].r, mid, r, x);
            }
        }
        return p;
    }
    // 向序列末尾追加离散秩 x，生成新的前缀频数版本；要求 1<=x<=m，O(log m)。
    void append(int x) {
        root.push_back(insert(root.back(), 1, m + 1, x));
    }
    // 返回原序列闭区间 [l,r] 的第 k 小离散秩，需再查离散值表取得实际值。
    // 要求 1<=l<=r<=已追加数量、1<=k<=r-l+1；时间 O(log m)。
    int kth(int l, int r, int k) {
        assert(1 <= k && k <= r - l + 1);
        int a = root[l - 1], b = root[r], L = 1, R = m + 1;
        while (R - L > 1) {
            int mid = (L + R) / 2, c = t[t[b].l].cnt - t[t[a].l].cnt;
            if (k <= c) {
                a = t[a].l;
                b = t[b].l;
                R = mid;
            }
            else {
                k -= c;
                a = t[a].r;
                b = t[b].r;
                L = mid;
            }
        }
        return L;
    }
};
```

**调用与说明**：依次 append(rank[i])（i=1..n），xs[tree.kth(l,r,k)] 即闭区间 [l,r] 第 k 小的实际值。

## 2.7 异或线性基

[完整代码：2.7 异或线性基.cpp](../%E4%BB%A3%E7%A0%81/02_%E6%95%B0%E6%8D%AE%E7%BB%93%E6%9E%84/2.7%20%E5%BC%82%E6%88%96%E7%BA%BF%E6%80%A7%E5%9F%BA.cpp)

**用途与条件**：维护 64 位无符号整数的异或空间，支持插入、判断可表示性、最大异或和。 例外：二进制位编号为 0..63，基数组 b[i] 按位编号。

**复杂度**：每次操作 O(64)，空间 O(64)。

**易错点**：本模板允许选择空集。第 k 小异或值需要进一步消元及明确是否计入空集，不能直接把基按大小排列。

```cpp
struct XorBasis {
    // b[i]：最高有效位为 i 的基向量，0 表示该位置尚无基向量。
    // XOR（Exclusive OR）：按位异或。
    array<u64, 64> b{};
    // 将 x 加入异或线性基；使线性空间维数增加时返回 true，否则返回 false。
    // 会修改基向量，处理完整 64 位无符号范围，时间 O(64)。
    bool insert(u64 x) {
        for (int i = 63; i >= 0; --i) {
            if ((x >> i) & 1) {
                if (!b[i]) {
                    b[i] = x;
                    return true;
                }
                x ^= b[i];
            }
        }
        return false;
    }
    // 判断 x 能否由已插入数的某个子集异或得到；允许空集，因此 0 总可表示，O(64)。
    bool contains(u64 x) const {
        for (int i = 63; i >= 0; --i) {
            if ((x >> i) & 1) {
                x ^= b[i];
            }
        }
        return x == 0;
    }
    // 返回 x 与线性空间中某个值异或后的最大无符号结果；默认 x=0 求最大子集异或和。
    // 不修改线性基，允许选择空集，时间 O(64)。
    u64 maximum(u64 x = 0) const {
        for (int i = 63; i >= 0; --i) {
            x = max(x, x ^ b[i]);
        }
        return x;
    }
};
```

**调用与说明**：插入 3 和 5 后，maximum()==6，contains(6)==true。

## 2.8 莫队：区间不同数个数

[完整代码：2.8 莫队：区间不同数个数.cpp](../%E4%BB%A3%E7%A0%81/02_%E6%95%B0%E6%8D%AE%E7%BB%93%E6%9E%84/2.8%20%E8%8E%AB%E9%98%9F%EF%BC%9A%E5%8C%BA%E9%97%B4%E4%B8%8D%E5%90%8C%E6%95%B0%E4%B8%AA%E6%95%B0.cpp)

**用途与条件**：输入 a[1..n]，值离散化为 1..m。ranges 是无占位的查询列表，每项是闭区间 [l,r]；返回答案按查询列表顺序存放（列表位置从 0 开始）。

**复杂度**：块长约 sqrt(n)，时间 O((n+q)sqrt(n))，空间 O(n+m+q)。

**易错点**：本模板不支持修改。答案按原查询顺序返回。add/remove 必须互为逆操作；奇偶块排序降低常数。

```cpp
// 离线统计 ranges 中各闭区间的不同值个数，答案按查询输入顺序返回，不带占位。
// a 使用 1 下标且值在 1..m；非空查询满足 1<=l<=r<=n，不支持修改。
vector<int> mo_distinct(const vector<int>& a, int m, const vector<pair<int, int>>& ranges) {
    int n = (int)a.size() - 1, q = ranges.size(), B = max(1, (int)sqrt(n));
    // ord：查询的处理顺序；ans：按原查询编号保存答案；cnt[x]：当前区间中值 x 的频数。
    vector<int> ord(q), ans(q), cnt(m + 1);
    iota(ord.begin(), ord.end(), 0);
    // 按左端点分块，块内右端点交替升降序排序，减少窗口指针移动；x、y 是查询编号。
    sort(ord.begin(), ord.end(), [&](int x, int y) {
        auto [l, r] = ranges[x];
        auto [L, R] = ranges[y];
        if (l / B != L / B) return l / B < L / B;
        return (l / B & 1) ? r > R : r < R;
    });
    // 当前窗口为 [l,r]，初始为空；cur：窗口内不同数值的数量。
    int l = 1, r = 0, cur = 0;
    // 把位置 i 纳入当前窗口，更新其频数；该值首次出现时增加不同值计数 cur。
    auto add = [&](int i) {
        if (cnt[a[i]]++ == 0) ++cur;
    };
    // 把当前窗口内的位置 i 移出，更新其频数；该值不再出现时减少 cur。
    auto del = [&](int i) {
        if (--cnt[a[i]] == 0) --cur;
    };
    for (int id : ord) {
        auto [L, R] = ranges[id];
        while (l > L) add(--l);
        while (r < R) add(++r);
        while (l < L) del(l++);
        while (r > R) del(r--);
        ans[id] = cur;
    }
    return ans;
}
```

**调用与说明**：mo_distinct({0,1,2,1},2,{{1,3},{2,2}}) 返回 {2,1}。

## 2.9 隐式 FHQ Treap：序列翻转

[完整代码：2.9 隐式 FHQ Treap：序列翻转.cpp](../%E4%BB%A3%E7%A0%81/02_%E6%95%B0%E6%8D%AE%E7%BB%93%E6%9E%84/2.9%20%E9%9A%90%E5%BC%8F%20FHQ%20Treap%EF%BC%9A%E5%BA%8F%E5%88%97%E7%BF%BB%E8%BD%AC.cpp)

**用途与条件**：序列位置为 1..n，reverse(l,r) 翻转闭区间 [l,r]，要求 l≤r。values() 返回带下标 0 占位的数组。split 的 k 仍表示元素个数，节点 0 为空。

**复杂度**：期望单次 O(log n)，空间 O(n)；随机优先级下深度期望 O(log n)。

**易错点**：split 按元素个数分割。访问儿子前 push。合并要求 a 的全部元素在 b 之前。本实现不回收删除节点。

```cpp
// FHQ Treap：基于 split/merge 的无旋 Treap；FHQ 来自作者名 fhq。
// Treap 由 Tree（树）和 Heap（堆）合成；隐式版本用中序位置表示序列下标。
struct ImplicitTreap {
    struct Node {
        // l、r：左右儿子编号，0 表示空树；sz：以当前节点为根的子树大小。
        int l = 0, r = 0, sz = 1;
        // pri（priority）：随机优先级，维护堆性质。
        u64 pri = 0;
        i64 val = 0;
        // rev（reverse）：待下传的区间翻转标记，不是反向下标。
        bool rev = false;
    };
    // t：节点池，0 号为空节点；有效节点从 1 开始。
    vector<Node> t{{}};
    int root = 0;
    mt19937_64 rng{712367};
    // 返回节点 p 的子树大小；节点编号 0 表示空树，返回 0，时间 O(1)。
    int size(int p) { return p ? t[p].sz : 0; }
    // 内部维护：根据左右孩子重新计算非空节点 p 的子树大小，时间 O(1)。
    void pull(int p) {
        t[p].sz = 1 + size(t[p].l) + size(t[p].r);
    }
    // 为子树 p 应用翻转：交换左右孩子并切换延迟翻转标记；空树不操作，O(1)。
    void flip(int p) {
        if (p) {
            swap(t[p].l, t[p].r);
            t[p].rev ^= 1;
        }
    }
    // 将非空节点 p 的延迟翻转标记下传给孩子，并清除自身标记，时间 O(1)。
    void push(int p) {
        if (t[p].rev) {
            flip(t[p].l);
            flip(t[p].r);
            t[p].rev = 0;
        }
    }
    // 按元素个数把子树 p 分为前 k 项与剩余项，返回两棵树的根；要求 0<=k<=size(p)。
    // 会重接子树并下传标记，期望时间 O(log n)。
    pair<int, int> split(int p, int k) {
        if (!p) return {0, 0};
        push(p);
        if (size(t[p].l) >= k) {
            auto [a, b] = split(t[p].l, k);
            t[p].l = b;
            pull(p);
            return {a, p};
        }
        auto [a, b] = split(t[p].r, k - size(t[p].l) - 1);
        t[p].r = a;
        pull(p);
        return {p, b};
    }
    // 把序列树 a 接在 b 前面合并，返回新根；要求 a 的所有项排在 b 的所有项之前。
    // 会修改树的连接关系，期望时间 O(log n)。
    int merge(int a, int b) {
        if (!a || !b) return a ? a : b;
        if (t[a].pri > t[b].pri) {
            push(a);
            t[a].r = merge(t[a].r, b);
            pull(a);
            return a;
        }
        push(b);
        t[b].l = merge(a, t[b].l);
        pull(b);
        return b;
    }
    // 在当前序列末尾添加值 v，并更新 root；期望时间 O(log n)，新增一个节点。
    void append(i64 v) {
        int p = t.size();
        Node x;
        x.pri = rng();
        x.val = v;
        t.push_back(x);
        root = merge(root, p);
    }
    // 原地翻转序列的闭区间 [l,r]，要求 1<=l<=r<=序列长度；期望时间 O(log n)。
    void reverse(int l, int r) {
        auto [a, c] = split(root, r);
        auto [x, b] = split(a, l - 1);
        flip(b);
        root = merge(merge(x, b), c);
    }
    // 内部中序遍历：把子树 p 的值依次追加到 a，遍历时下传翻转标记。
    // 不清空 a；耗时与子树大小成正比，递归空间与树高成正比。
    void collect(int p, vector<i64>& a) {
        if (!p) return;
        push(p);
        collect(t[p].l, a);
        a.push_back(t[p].val);
        collect(t[p].r, a);
    }
    // 导出当前完整序列，返回 a[1..n]，a[0] 为占位；时间和输出空间 O(n)。
    vector<i64> values() {
        vector<i64> a(1);
        collect(root, a);
        return a;
    }
};
```

**调用与说明**：依次 append(1),append(2),append(3)，reverse(1,2) 后 values() 为 {0,2,1,3}。

## 2.10 01 Trie：二进制字典树

[完整代码：2.10 01 Trie：二进制字典树.cpp](../%E4%BB%A3%E7%A0%81/02_%E6%95%B0%E6%8D%AE%E7%BB%93%E6%9E%84/2.10%2001%20Trie%EF%BC%9A%E4%BA%8C%E8%BF%9B%E5%88%B6%E5%AD%97%E5%85%B8%E6%A0%91.cpp)

**用途与条件**：维护非负整数多重集合，支持插入、删除一个数、计数和最大异或查询。构造参数 max_bit 是最高位编号（0..63），默认 63；节点从 1 开始，根为 1，0 表示空子节点。二进制位从 0 开始编号。

**复杂度**：插入、删除、计数和最大异或查询均为 O(max_bit+1)，size 为 O(1)。空间 O(1+历史插入次数×(max_bit+1))；clear 清空节点但不保证释放容量。

**易错点**：所有操作的参数 x 必须在设定的位宽内；max_bit=63 时支持完整 u64 范围。max_xor 要求集合非空，返回异或值而非选中的数。删除只减少计数，不回收节点，查询必须跳过计数为 0 的分支。节点编号和计数使用 int，规模须在其范围内。

```cpp
// 01 Trie（Binary Trie）：二进制字典树，支持重复数。
class BinaryTrie {
private:
    struct Node {
        int ch[2] = {};  // 当前位为 0、1 的子节点，0 表示不存在
        int cnt = 0;    // 经过该节点的数的数量，包含重复数
    };

    int max_bit;      // 最高二进制位编号，共处理 max_bit+1 位
    vector<Node> tr;  // tr[0] 占位，tr[1] 为根；删除后保留节点

public:
    // 建立空字典树；最高位编号 max_bit 必须在 0..63 内。
    BinaryTrie(int max_bit = 63) : max_bit(max_bit), tr(2) {
        assert(0 <= max_bit && max_bit <= 63);
    }

    // 清空所有数并恢复根节点，保留设定的最高位编号。
    void clear() {
        tr.assign(2, Node{});
    }

    // 返回当前数的数量，包含重复数。
    int size() const {
        return tr[1].cnt;
    }

    // 返回 x 的出现次数；x 必须在设定位宽内，不存在时返回 0。
    int count(u64 x) const {
        assert(max_bit == 63 || (x >> (max_bit + 1)) == 0);
        int u = 1;
        for (int i = max_bit; i >= 0; i--) {
            int bit = (x >> i) & 1ULL;
            int v = tr[u].ch[bit];
            if (v == 0 || tr[v].cnt == 0) return 0;
            u = v;
        }
        return tr[u].cnt;
    }

    // 插入位宽范围内的 x，允许重复插入。
    void insert(u64 x) {
        assert(max_bit == 63 || (x >> (max_bit + 1)) == 0);
        int u = 1;
        tr[u].cnt++;
        for (int i = max_bit; i >= 0; i--) {
            int bit = (x >> i) & 1ULL;
            if (tr[u].ch[bit] == 0) {
                int v = (int)tr.size();
                tr.emplace_back();
                tr[u].ch[bit] = v;
            }
            u = tr[u].ch[bit];
            tr[u].cnt++;
        }
    }

    // 删除位宽范围内的一个 x；成功返回 true，不存在则返回 false。
    bool erase(u64 x) {
        if (count(x) == 0) return false;
        int u = 1;
        tr[u].cnt--;
        for (int i = max_bit; i >= 0; i--) {
            int bit = (x >> i) & 1ULL;
            u = tr[u].ch[bit];
            tr[u].cnt--;
        }
        return true;
    }

    // 返回集合中各个 y 与 x 异或后的最大值，不是 y 本身。
    // 集合必须非空，x 必须在设定位宽内。
    u64 max_xor(u64 x) const {
        assert(size() > 0);
        assert(max_bit == 63 || (x >> (max_bit + 1)) == 0);
        int u = 1;
        u64 ans = 0;
        for (int i = max_bit; i >= 0; i--) {
            int bit = (x >> i) & 1ULL;
            int v = tr[u].ch[bit ^ 1];
            // 高位优先选择相反位，使异或结果尽量大；跳过已删除的分支。
            if (v != 0 && tr[v].cnt > 0) {
                ans |= 1ULL << i;
                u = v;
            }
            else {
                u = tr[u].ch[bit];
            }
        }
        return ans;
    }
};
```

**调用与说明**：BinaryTrie trie(30); 处理第 30..0 位，要求 x<2^31。依次 insert(5)、insert(5)、insert(3)，erase(5) 后 count(5)==1、max_xor(2)==7；再 erase(5) 后 max_xor(2)==1；再次 erase(5) 返回 false。默认 BinaryTrie trie; 支持完整 64 位范围。
